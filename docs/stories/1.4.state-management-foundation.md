## Story

**As a** developer,
**I want** to implement robust state management for the Next.js application,
**so that** user actions are tracked, the app remains responsive and scalable, and provides undo/redo functionality critical for creative tools.

## Acceptance Criteria

1. Zustand store configured with TypeScript interfaces for complete type safety in Next.js app
2. Store organized by domain slices (canvas, upload, ui, history) compatible with Next.js patterns
3. Each slice properly typed with state and action interfaces using Next.js client components
4. Store middleware configured for development debugging with Next.js DevTools integration
5. Current session state automatically saved to localStorage on changes with Next.js client-side hydration
6. State restoration on application reload preserves user work in Next.js SSG environment
7. Action history automatically tracked for undoable operations in Next.js client components
8. Undo (Ctrl/Cmd+Z) and Redo (Ctrl/Cmd+Shift+Z) keyboard shortcuts functional
9. State updates trigger only necessary component re-renders optimized for Next.js
10. Clear state action available for starting fresh projects in Next.js app

```typescript
// stores/index.ts (Next.js App Router)
'use client'

import { create } from 'zustand'
import { subscribeWithSelector } from 'zustand/middleware'
import { persist } from 'zustand/middleware'
import { immer } from 'zustand/middleware/immer'
import { devtools } from 'zustand/middleware'

// Store slices for Next.js
import { createCanvasSlice, CanvasSlice } from './canvas-slice'
import { createUploadSlice, UploadSlice } from './upload-slice'
import { createHistorySlice, HistorySlice } from './history-slice'
import { createUISlice, UISlice } from './ui-slice'

// Combined store type for Next.js
export interface AppStore extends CanvasSlice, UploadSlice, HistorySlice, UISlice {}

// Create store with middleware optimized for Next.js
export const useAppStore = create<AppStore>()(
  devtools(
    subscribeWithSelector(
      persist(
        immer((set, get, api) => ({
          ...createCanvasSlice(set, get, api),
          ...createUploadSlice(set, get, api),
          ...createHistorySlice(set, get, api),
          ...createUISlice(set, get, api),
        })),
        {
          name: 'scroma-app-storage',
          version: 1,
          partialize: (state) => ({
            canvas: state.canvas,
            upload: {
              recentFiles: state.upload.recentFiles,
              settings: state.upload.settings,
            },
            ui: {
              preferences: state.ui.preferences,
              panelStates: state.ui.panelStates,
            },
            // Exclude history and temporary state from persistence for Next.js SSG
          }),
          onRehydrateStorage: () => (state) => {
            // Initialize non-persisted state after rehydration in Next.js
            state?.initializeSession()
          },
        }
      )
    ),
    { name: 'Scroma App Store - Next.js' }
  )
)
```

**Canvas Slice Implementation**:
```typescript
// stores/canvas-slice.ts (Next.js App Router)
'use client'

import { StateCreator } from 'zustand'
import { AppStore } from './index'

export interface CanvasState {
  image: {
    src: string | null
    originalWidth: number
    originalHeight: number
    fileName: string
    fileSize: number
    format: string
  } | null

  viewport: {
    width: number
    height: number
    zoom: number
    panX: number
    panY: number
    minZoom: number
    maxZoom: number
  }

  interaction: {
    isDragging: boolean
    isZooming: boolean
    dragStart: { x: number; y: number } | null
  }
}

export interface CanvasActions {
  setImage: (imageData: ImageData) => void
  clearImage: () => void
  setZoom: (zoom: number) => void
  setPan: (x: number, y: number) => void
  setViewportSize: (width: number, height: number) => void
  resetCanvas: () => void
  startDrag: (x: number, y: number) => void
  updateDrag: (x: number, y: number) => void
  endDrag: () => void
}

export interface CanvasSlice extends CanvasState, CanvasActions {}

const initialCanvasState: CanvasState = {
  image: null,
  viewport: {
    width: parseInt(process.env.NEXT_PUBLIC_DEFAULT_CANVAS_WIDTH || '1200'),
    height: parseInt(process.env.NEXT_PUBLIC_DEFAULT_CANVAS_HEIGHT || '800'),
    zoom: 1,
    panX: 0,
    panY: 0,
    minZoom: parseFloat(process.env.NEXT_PUBLIC_MIN_ZOOM || '0.1'),
    maxZoom: parseFloat(process.env.NEXT_PUBLIC_MAX_ZOOM || '5.0'),
  },
  interaction: {
    isDragging: false,
    isZooming: false,
    dragStart: null,
  },
}

export const createCanvasSlice: StateCreator<
  AppStore,
  [["zustand/immer", never]],
  [],
  CanvasSlice
> = (set, get) => ({
  ...initialCanvasState,

  setImage: (imageData) =>
    set((state) => {
      state.canvas.image = imageData
      // Record action in history for Next.js
      state.history.addAction({
        type: 'SET_IMAGE',
        timestamp: Date.now(),
        data: { imageData },
      })
    }),

  clearImage: () =>
    set((state) => {
      const previousImage = state.canvas.image
      state.canvas.image = null
      state.canvas.viewport.zoom = 1
      state.canvas.viewport.panX = 0
      state.canvas.viewport.panY = 0
      // Record action in history with previous state
      state.history.addAction({
        type: 'CLEAR_IMAGE',
        timestamp: Date.now(),
        data: { previousImage },
      })
    }),

  setZoom: (zoom) =>
    set((state) => {
      const clampedZoom = Math.max(
        state.canvas.viewport.minZoom,
        Math.min(state.canvas.viewport.maxZoom, zoom)
      )
      state.canvas.viewport.zoom = clampedZoom
    }),

  setPan: (x, y) =>
    set((state) => {
      state.canvas.viewport.panX = x
      state.canvas.viewport.panY = y
    }),

  setViewportSize: (width, height) =>
    set((state) => {
      state.canvas.viewport.width = width
      state.canvas.viewport.height = height
    }),

  resetCanvas: () =>
    set((state) => {
      state.canvas.viewport.zoom = 1
      state.canvas.viewport.panX = 0
      state.canvas.viewport.panY = 0
      state.canvas.interaction = initialCanvasState.interaction
    }),

  startDrag: (x, y) =>
    set((state) => {
      state.canvas.interaction.isDragging = true
      state.canvas.interaction.dragStart = { x, y }
    }),

  updateDrag: (x, y) =>
    set((state) => {
      if (state.canvas.interaction.isDragging && state.canvas.interaction.dragStart) {
        const deltaX = x - state.canvas.interaction.dragStart.x
        const deltaY = y - state.canvas.interaction.dragStart.y
        state.canvas.viewport.panX += deltaX
        state.canvas.viewport.panY += deltaY
        state.canvas.interaction.dragStart = { x, y }
      }
    }),

  endDrag: () =>
    set((state) => {
      state.canvas.interaction.isDragging = false
      state.canvas.interaction.dragStart = null
    }),
})
```

```typescript
// stores/upload-slice.ts
'use client'

export interface UploadState {
  files: UploadFile[]
  currentUpload: string | null
  isUploading: boolean
  uploadProgress: number
  queue: string[]
  settings: {
    maxFileSize: number
    supportedFormats: string[]
    autoProcess: boolean
  }
  recentFiles: RecentFile[]
}

export interface UploadActions {
  addFile: (file: File) => void
  removeFile: (fileId: string) => void
  updateProgress: (fileId: string, progress: number) => void
  setUploadStatus: (fileId: string, status: UploadStatus) => void
  clearCompleted: () => void
  updateSettings: (settings: Partial<UploadSettings>) => void
}

export interface UploadSlice extends UploadState, UploadActions {}

const initialUploadState: UploadState = {
  files: [],
  currentUpload: null,
  isUploading: false,
  uploadProgress: 0,
  queue: [],
  settings: {
    maxFileSize: parseInt(process.env.NEXT_PUBLIC_MAX_FILE_SIZE || '10485760'),
    supportedFormats: process.env.NEXT_PUBLIC_SUPPORTED_FORMATS?.split(',') || ['png', 'jpg', 'webp'],
    autoProcess: true,
  },
  recentFiles: [],
}

export const createUploadSlice: StateCreator<AppStore, [["zustand/immer", never]], [], UploadSlice> = (set, get) => ({
  ...initialUploadState,

  addFile: (file) =>
    set((state) => {
      const uploadFile: UploadFile = {
        id: crypto.randomUUID(),
        file,
        status: 'pending',
        progress: 0,
        addedAt: Date.now(),
      }
      state.upload.files.push(uploadFile)
      state.upload.queue.push(uploadFile.id)
    }),

  removeFile: (fileId) =>
    set((state) => {
      state.upload.files = state.upload.files.filter(f => f.id !== fileId)
      state.upload.queue = state.upload.queue.filter(id => id !== fileId)
    }),

  updateProgress: (fileId, progress) =>
    set((state) => {
      const file = state.upload.files.find(f => f.id === fileId)
      if (file) {
        file.progress = progress
      }
    }),

  setUploadStatus: (fileId, status) =>
    set((state) => {
      const file = state.upload.files.find(f => f.id === fileId)
      if (file) {
        file.status = status
        if (status === 'completed') {
          // Add to recent files
          state.upload.recentFiles.unshift({
            name: file.file.name,
            size: file.file.size,
            completedAt: Date.now(),
          })
          // Keep only last 10 recent files
          state.upload.recentFiles = state.upload.recentFiles.slice(0, 10)
        }
      }
    }),
})
```

**History Slice Implementation** [Source: story 1.4 existing content]:

```typescript
// stores/history-slice.ts (Next.js App Router)
'use client'

export interface HistoryAction {
  type: string
  timestamp: number
  data: any
  description?: string
}

export interface HistoryState {
  actions: HistoryAction[]
  currentIndex: number
  maxSize: number
  canUndo: boolean
  canRedo: boolean
}

export interface HistoryActions {
  addAction: (action: HistoryAction) => void
  undo: () => void
  redo: () => void
  clearHistory: () => void
  canPerformUndo: () => boolean
  canPerformRedo: () => boolean
}

export interface HistorySlice extends HistoryState, HistoryActions {}

const initialHistoryState: HistoryState = {
  actions: [],
  currentIndex: -1,
  maxSize: parseInt(process.env.NEXT_PUBLIC_MAX_UNDO_HISTORY || '50'),
  canUndo: false,
  canRedo: false,
}

export const createHistorySlice: StateCreator<AppStore, [["zustand/immer", never]], [], HistorySlice> = (set, get) => ({
  ...initialHistoryState,

  addAction: (action) =>
    set((state) => {
      // Remove any actions after current index (if we're in middle of history)
      state.history.actions = state.history.actions.slice(0, state.history.currentIndex + 1)

      // Add new action
      state.history.actions.push(action)
      state.history.currentIndex = state.history.actions.length - 1

      // Trim history if it exceeds max size (Next.js memory optimization)
      if (state.history.actions.length > state.history.maxSize) {
        state.history.actions = state.history.actions.slice(-state.history.maxSize)
        state.history.currentIndex = state.history.actions.length - 1
      }

      // Update undo/redo flags for Next.js components
      state.history.canUndo = state.history.currentIndex >= 0
      state.history.canRedo = state.history.currentIndex < state.history.actions.length - 1
    }),

  undo: () =>
    set((state) => {
      if (state.history.currentIndex >= 0) {
        const action = state.history.actions[state.history.currentIndex]

        // Apply undo logic based on action type for Next.js
        switch (action.type) {
          case 'SET_IMAGE':
            state.canvas.image = null
            break
          case 'CLEAR_IMAGE':
            state.canvas.image = action.data.previousImage
            break
          // Add more undo cases as needed for Next.js
        }

        state.history.currentIndex -= 1
        state.history.canUndo = state.history.currentIndex >= 0
        state.history.canRedo = true
      }
    }),

  redo: () =>
    set((state) => {
      if (state.history.currentIndex < state.history.actions.length - 1) {
        state.history.currentIndex += 1
        const action = state.history.actions[state.history.currentIndex]

        // Apply redo logic based on action type for Next.js
        switch (action.type) {
          case 'SET_IMAGE':
            state.canvas.image = action.data.imageData
            break
          case 'CLEAR_IMAGE':
            state.canvas.image = null
            break
          // Add more redo cases as needed for Next.js
        }

        state.history.canRedo = state.history.currentIndex < state.history.actions.length - 1
        state.history.canUndo = true
      }
    }),

  clearHistory: () =>
    set((state) => {
      state.history.actions = []
      state.history.currentIndex = -1
      state.history.canUndo = false
      state.history.canRedo = false
    }),

  canPerformUndo: () => get().history.canUndo,
  canPerformRedo: () => get().history.canRedo,
})
```

**Keyboard Shortcuts Implementation**:
```typescript
// hooks/use-keyboard-shortcuts.ts
'use client'

export function useKeyboardShortcuts() {
  const { undo, redo, canUndo, canRedo } = useAppStore(
    useCallback((state) => ({
      undo: state.undo,
      redo: state.redo,
      canUndo: state.canUndo,
      canRedo: state.canRedo,
    }), [])
  )

  useEffect(() => {
    const handleKeydown = (event: KeyboardEvent) => {
      if (event.ctrlKey || event.metaKey) {
        if (event.shiftKey && event.key === 'Z' && canRedo) {
          event.preventDefault()
          redo()
        } else if (event.key === 'z' && canUndo) {
          event.preventDefault()
          undo()
        }
      }
    }

    document.addEventListener('keydown', handleKeydown)
    return () => document.removeEventListener('keydown', handleKeydown)
  }, [undo, redo, canUndo, canRedo])
}
```
