# Story 1.3: Canvas Display System

## Status

Draft

## Story

**As a** user,
**I want** to view my uploaded images on an interactive canvas with zoom and pan controls,
**so that** I can see my screenshots clearly and navigate around them for detailed editing.

## Acceptance Criteria

1. Uploaded image displays centered on canvas with appropriate initial zoom
2. Canvas dimensions adjust to viewport while maintaining aspect ratio
3. Zoom controls (+/-, slider, fit-to-screen) functional with 10-500% range
4. Pan functionality via drag when zoomed in beyond viewport
5. Image metadata (dimensions, file size) displayed in status bar
6. Canvas has subtle background pattern to indicate transparent areas
7. Reset view button returns to initial centered state
8. Canvas performance optimized for Next.js SSG and 60fps interactions

**Canvas Viewport Component** [Source: ui-architecture/frontend-developer-standards.md#86]:
```typescript
// components/canvas/canvas-viewport.tsx
'use client'

export function CanvasViewport({ imageData }: { imageData: ImageData | null }) {
  const canvasRef = useRef<HTMLCanvasElement>(null)
  const [viewport, setViewport] = useState({
    width: 1200,
    height: 800,
    zoom: 1,
    panX: 0,
    panY: 0
  })

  useEffect(() => {
    if (!canvasRef.current || !imageData) return

    const renderFrame = () => {
      const ctx = canvasRef.current?.getContext('2d')
      if (!ctx) return

      // Clear canvas
      ctx.clearRect(0, 0, viewport.width, viewport.height)

      // Apply transform matrix
      ctx.save()
      ctx.translate(viewport.panX, viewport.panY)
      ctx.scale(viewport.zoom, viewport.zoom)

      // Draw image
      ctx.drawImage(imageData.image, 0, 0)
      ctx.restore()
    }

    requestAnimationFrame(renderFrame)
  }, [imageData, viewport])

  return (
    <canvas
      ref={canvasRef}
      width={viewport.width}
      height={viewport.height}
      className="canvas-viewport"
    />
  )
}
```

**Canvas Renderer Engine** [Source: ui-architecture/project-structure.md#79]:
```typescript
// lib/canvas/canvas-renderer.ts
export class CanvasRenderer {
  private canvas: HTMLCanvasElement
  private ctx: CanvasRenderingContext2D
  private devicePixelRatio: number

  constructor(canvas: HTMLCanvasElement) {
    this.canvas = canvas
    this.ctx = canvas.getContext('2d')!
    this.devicePixelRatio = window.devicePixelRatio || 1
    this.setupCanvas()
  }

  private setupCanvas() {
    const rect = this.canvas.getBoundingClientRect()

    // Handle high-DPI displays
    this.canvas.width = rect.width * this.devicePixelRatio
    this.canvas.height = rect.height * this.devicePixelRatio
    this.canvas.style.width = rect.width + 'px'
    this.canvas.style.height = rect.height + 'px'

    this.ctx.scale(this.devicePixelRatio, this.devicePixelRatio)
  }

  public renderImage(image: HTMLImageElement, transform: Transform) {
    this.ctx.save()

    // Apply transform matrix
    this.ctx.translate(transform.panX, transform.panY)
    this.ctx.scale(transform.zoom, transform.zoom)

    // Enable image smoothing for better quality
    this.ctx.imageSmoothingEnabled = true
    this.ctx.imageSmoothingQuality = 'high'

    // Center image on canvas
    const x = (this.canvas.width / this.devicePixelRatio - image.width) / 2
    const y = (this.canvas.height / this.devicePixelRatio - image.height) / 2

    this.ctx.drawImage(image, x, y)
    this.ctx.restore()
  }

  public renderBackground(pattern: BackgroundPattern) {
    if (pattern.type === 'grid') {
      this.renderGridPattern(pattern)
    }
  }

  private renderGridPattern(pattern: GridPattern) {
    const gridSize = pattern.size || 20
    this.ctx.strokeStyle = pattern.color || '#e5e7eb'
    this.ctx.lineWidth = 1

    // Draw grid lines
    for (let x = 0; x <= this.canvas.width; x += gridSize) {
      this.ctx.beginPath()
      this.ctx.moveTo(x, 0)
      this.ctx.lineTo(x, this.canvas.height)
      this.ctx.stroke()
    }

    for (let y = 0; y <= this.canvas.height; y += gridSize) {
      this.ctx.beginPath()
      this.ctx.moveTo(0, y)
      this.ctx.lineTo(this.canvas.width, y)
      this.ctx.stroke()
    }
  }
}
```

**Zoom Controls Implementation**:
```typescript
// components/canvas/zoom-controls.tsx
'use client'

interface ZoomControlsProps {
  zoom: number
  onZoomChange: (zoom: number) => void
  onFitToScreen: () => void
  onResetView: () => void
}

export function ZoomControls({ zoom, onZoomChange, onFitToScreen, onResetView }: ZoomControlsProps) {
  const config = {
    minZoom: parseFloat(process.env.NEXT_PUBLIC_MIN_ZOOM || '0.1'),
    maxZoom: parseFloat(process.env.NEXT_PUBLIC_MAX_ZOOM || '5.0'),
    zoomStep: parseFloat(process.env.NEXT_PUBLIC_ZOOM_STEP || '0.2')
  }

  const handleZoomIn = () => {
    const newZoom = Math.min(config.maxZoom, zoom + config.zoomStep)
    onZoomChange(newZoom)
  }

  const handleZoomOut = () => {
    const newZoom = Math.max(config.minZoom, zoom - config.zoomStep)
    onZoomChange(newZoom)
  }

  const handleSliderChange = (value: number[]) => {
    onZoomChange(value[0])
  }

  return (
    <div className="zoom-controls">
      <Button
        variant="outline"
        size="sm"
        onClick={handleZoomOut}
        disabled={zoom <= config.minZoom}
        aria-label="Zoom out"
      >
        <Minus className="h-4 w-4" />
      </Button>

      <Slider
        value={[zoom]}
        onValueChange={handleSliderChange}
        min={config.minZoom}
        max={config.maxZoom}
        step={0.1}
        className="zoom-slider"
      />

      <Button
        variant="outline"
        size="sm"
        onClick={handleZoomIn}
        disabled={zoom >= config.maxZoom}
        aria-label="Zoom in"
      >
        <Plus className="h-4 w-4" />
      </Button>

      <span className="zoom-percentage">
        {Math.round(zoom * 100)}%
      </span>

      <Button
        variant="outline"
        size="sm"
        onClick={onFitToScreen}
        aria-label="Fit to screen"
      >
        <Maximize className="h-4 w-4" />
      </Button>

      <Button
        variant="outline"
        size="sm"
        onClick={onResetView}
        aria-label="Reset view"
      >
        <RotateCcw className="h-4 w-4" />
      </Button>
    </div>
  )
}
```

**Pan Functionality**:
```typescript
// hooks/use-canvas-state.ts
'use client'

export function useCanvasState(initialImage?: HTMLImageElement) {
  const [viewport, setViewport] = useState({
    width: parseInt(process.env.NEXT_PUBLIC_DEFAULT_CANVAS_WIDTH || '1200'),
    height: parseInt(process.env.NEXT_PUBLIC_DEFAULT_CANVAS_HEIGHT || '800'),
    zoom: 1,
    panX: 0,
    panY: 0
  })

  const [isDragging, setIsDragging] = useState(false)
  const [dragStart, setDragStart] = useState<{ x: number; y: number } | null>(null)

  const handleMouseDown = useCallback((event: React.MouseEvent) => {
    if (viewport.zoom > 1) { // Only allow panning when zoomed in
      setIsDragging(true)
      setDragStart({ x: event.clientX, y: event.clientY })
    }
  }, [viewport.zoom])

  const handleMouseMove = useCallback((event: React.MouseEvent) => {
    if (isDragging && dragStart) {
      const deltaX = event.clientX - dragStart.x
      const deltaY = event.clientY - dragStart.y

      setViewport(prev => ({
        ...prev,
        panX: prev.panX + deltaX,
        panY: prev.panY + deltaY
      }))

      setDragStart({ x: event.clientX, y: event.clientY })
    }
  }, [isDragging, dragStart])

  const handleMouseUp = useCallback(() => {
    setIsDragging(false)
    setDragStart(null)
  }, [])

  const handleWheel = useCallback((event: WheelEvent) => {
    event.preventDefault()

    const delta = event.deltaY > 0 ? -0.1 : 0.1
    const newZoom = Math.max(0.1, Math.min(5.0, viewport.zoom + delta))

    // Zoom towards mouse position
    const rect = (event.target as Element).getBoundingClientRect()
    const centerX = event.clientX - rect.left
    const centerY = event.clientY - rect.top

    setViewport(prev => ({
      ...prev,
      zoom: newZoom,
      panX: centerX - (centerX - prev.panX) * (newZoom / prev.zoom),
      panY: centerY - (centerY - prev.panY) * (newZoom / prev.zoom)
    }))
  }, [viewport.zoom])

  return {
    viewport,
    setViewport,
    isDragging,
    handlers: {
      onMouseDown: handleMouseDown,
      onMouseMove: handleMouseMove,
      onMouseUp: handleMouseUp,
      onWheel: handleWheel
    }
  }
}
```

### Performance Optimizations

**60fps Rendering** [Source: ui-architecture/frontend-developer-standards.md#62]:
- Use `requestAnimationFrame` for all canvas updates
- Implement render throttling with 16ms intervals
- Only redraw when necessary (viewport or image changes)
- Use transform matrices instead of recalculating positions

**Memory Management**:
- Properly dispose of ImageBitmap objects
- Clean up event listeners on component unmount
- Use canvas pooling for multiple instances
- Monitor memory usage in development mode
